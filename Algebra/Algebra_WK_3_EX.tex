\noindent\textbf{A)} Let $G$ be a group and let $H,K\subset G$ be subgroups of $G$. Show that if $G=H\cup K$ then $H=G$ or $K=G$.\\
(\textit{Hint:} Argue by contradiction. If $a\in H-K$ and $b\in K-H$, consider $ab$.)\\

\vspace{1cm}
\noindent
Answer:\\
\\
Given $H,K<G$ such that: $H\cup K=G$ (*).\\
\\
Assume that $H \not\subset K$ and $K \not\subset H$ then $\exists h\in H\setminus K$ and $k\in K\setminus H$.\\
Then:
\begin{align*}
hk&\in G=H\cup K.\\
\Rightarrow\quad hk&\in H\quad\text{or}\quad hk\in K.
\end{align*}
If we assume $hk\in H$.
\begin{align*}
\Rightarrow\quad hk&=h'\quad\text{for some }h'\in H.\\
\Rightarrow\quad k&=h^{-1}h'\in H.
\end{align*}
\underline{Contradiction} - which will also occuresif we assumn $hk \in K$\\
\\
Hence, if $(*)$ holds we must have $H < K$ or $K < H$.
\begin{align*}
\Rightarrow\quad H=G \quad\quad \text{OR \;\;\;\;$K=G$}
\end{align*}

\noindent\textbf{B)} Let $C_m$ and $C_n$ be two finite cyclic groups of cardinality $m$ and $n$ respectively. Let $d=\gcd(m,n)$ be the greatest common divisor of $m$ and $n$. We want to classify all morphisms of groups $f:C_m\to C_n$.

\begin{enumerate}
\item[(1)] Show that, for every $x\in C_m$, $\operatorname{ord}(f(x))$ is a divisor of $d$. (\textit{Hint:} Show that this number is both a divisor of $m$ and $n$.)
\item[(2)] Let $C_d\subset C_n$ be the unique subgroup of cardinality $d$. Show that the image of $f$ is a contained in $C_d$. What happens if $d=1$?
\item[(3)] Fix a generator $g\in C_m$. Show that for every $y\in C_d$, there is a unique morphism of groups $f:C_m\to C_n$ such that $f(g)=y$.
\item[(4)] What is the cardinality of the set of homomorphisms from $C_m$ to $C_n$?
\end{enumerate}

\vspace{1cm}
\noindent
Answer:\\
\\
(1) Let $x\in C_m\Rightarrow\operatorname{ord}(x)\mid m$.  But then $f(x)$ has $\operatorname{ord}[f(x)]\mid m$ since:
\begin{align*}
[f(x)]^m&=f(x^m)=f(1)\\
&=1
\end{align*}
But $f(x)\in C_n\Rightarrow\operatorname{ord}[f(x)]\mid n$.  Hence $\operatorname{ord}[f(x)]\mid n$ \& $\operatorname{ord}[f(x)]\mid m$
\begin{align*}
\Rightarrow\quad\operatorname{ord}[f(x)]\mid\gcd(m,n)
\end{align*}
\\
(2) Let $K=f(C_m)$.  Then every element of $K$ has order that divides $d=\gcd(m,n)$.
\begin{align*}
\therefore\quad K=f(C_m)\subset C_d \quad\quad\quad\quad \text{(since $C_d$ is unique)}
\end{align*}
If $d=1\Rightarrow C_1=\{1\}$ and $C_1$ only contains the identity element.\\
Hence
\begin{align*}
f(x)=1\qquad\forall x\in C_m.
\end{align*}
\\
(3) Let $\langle g\rangle=C_m$ and set $f(g)=y\in C_d$.\\
Then as $C_m$ is cyclic every element is of the form $g^k$ ($0\leq k<m$). Hence
\begin{align*}
f(g^k)=[f(g)]^k=y^k
\end{align*}
That is the map $f$ is defined uniquely from $C_m\to C_d$ (and hence for $C_m\to C_n$) for every element
in $C_m$.\\
\\
(4) The cardinality of the set of homomorphisms ($f: C_m \to C_n$) is defined by the number of different
maps that map $g$ (where $\langle g \rangle = C_m$) to an element of $C_n$ that has order $s$, where $s \mid gcd(m,n)$.

\noindent\textbf{C)} Let $S$ be a set. A map $p:S\to S$ is called a \textit{projector} if $p\circ p=p$. Let $G\subset\operatorname{Map}(S,S)$ be a subset closed under composition (i.e., if $f,g\in G$, then $f\circ g\in G$) and such that $(G,\circ)$ is a group. (Note that we are not asking $G$ to be a submonoid of $\operatorname{Map}(S,S)$, i.e., $G$ does not necessary contain $\operatorname{Id}_S$.)

\begin{enumerate}
\item[(1)] Show that $G$ contains one and only one projector. Let $p$ be the unique projector contained in $G$ and let $T=p(S)$ be its image. Let
\begin{align*}
q=p|_{S-T}:S-T\to T
\end{align*}
be the restriction of $p$ to $S-T$.

\item[(2)] Let $f\in G$. Show that $f(S)=T$ and that $f|_T:T\to T$ is a bijection. Show that $f|_{S-T}:S-T\to T$ is determined by $f|_T$ via the formula $f|_{S-T}=f|_T\circ q$. What is the inverse of $f$ in $G$?

\item[(3)] Show that given a projector $p:S\to S$, there exists a largest subset $G(p)\subset\operatorname{Map}(S,S)$, containing $p$, stable under composition, and such that $(G(p),\circ)$ is a group. Show that $G(p)$ is isomorphic to $(\operatorname{Bij}(T,T),\circ)$, where $T=p(S)$.
\end{enumerate}

\vspace{1cm}
\noindent
Answer:\\
\\
(1) $G$ is a group and $p \in G$.  We need to show that $p$ is the effective identity element
for $G$ and then its uniqueness is assured.  We do this as follows:
\begin{align*}
 p^{-1} \circ (p \circ p) &= p^{-1} \circ p \\
 &= e \quad \quad \quad \quad \text{(the identity for $G$)} \\
 &= (p^{-1} \circ p) \circ p \\
 &= 1 \circ p \\
 &= p
\end{align*}
Hence, $p$ acts as identity for $G$ and is therefore unique.
