\section*{Question 1 - \(\star\) Hamilton's Principle: The Euler--Lagrange equations and their invariances}
Let \(L=L(q,\dot q;t)\) be smooth and let
\begin{align*}
S[q]=\int_{t_0}^{t_1}L(q(t),\dot q(t);t)\,dt.
\end{align*}

\begin{enumerate}
\item[(a)] \textbf{(Fundamental lemma).}
Let \(F:[t_0,t_1]\to\mathbb{R}^f\) be continuous and suppose
\begin{align*}
\int_{t_0}^{t_1}F_a(t)\eta^a(t)\,dt=0
\end{align*}
for every smooth \(\eta\) with compact support in \((t_0,t_1)\).
Prove that \(F=0\).
\textit{Hint: if \(F_a(t_*)\neq 0\) for some \(t_*\) and some index \(a\),
construct a bump function concentrated near \(t_*\).}

\item[(b)] Compute the first variation
\begin{align*}
\delta S[q](\eta)
=
\left.
\frac{d}{d\epsilon}S[q+\epsilon\eta]
\right|_{\epsilon=0}
\end{align*}
for variations with \(\eta(t_0)=\eta(t_1)=0\), and deduce from (a) that
\(\delta S=0\) is equivalent to the Euler--Lagrange equations.

\item[(c)] Let \(F=F(q,t)\) be smooth and set
\begin{align*}
L'
=
L+\frac{d}{dt}F(q,t),
\qquad\text{i.e.}\qquad
L'
=
L+\frac{\partial F}{\partial q^a}\dot q^a
+\frac{\partial F}{\partial t}.
\end{align*}
Show that \(L\) and \(L'\) have exactly the same Euler--Lagrange equations.
How do the actions \(S\) and \(S'\) differ?

\item[(d)] \textbf{(Coordinate invariance).}
Let \(q=q(Q)\) be a diffeomorphic change of generalized coordinates and let
\begin{align*}
\widetilde L(Q,\dot Q;t)
=
L\left(q(Q),\frac{\partial q}{\partial Q}\dot Q;t\right).
\end{align*}
Prove the identity
\begin{align*}
\frac{d}{dt}\frac{\partial\widetilde L}{\partial \dot Q^b}
-
\frac{\partial\widetilde L}{\partial Q^b}
=
\frac{\partial q^a}{\partial Q^b}
\left(
\frac{d}{dt}\frac{\partial L}{\partial\dot q^a}
-
\frac{\partial L}{\partial q^a}
\right),
\end{align*}
and conclude that the Euler--Lagrange equations are coordinate independent.
Why does the Cartesian component form \(m\ddot x^a=F^a\) acquire connection
terms in curvilinear coordinates?
\end{enumerate}

(a) Let us assume that $F_a(t_*) > 0$ for some $t_*$ - it makes no difference if it would be negative in the argument.
As $F$ is continuous there is an interval $[t_* - \delta, t_* + \delta]$ for which $F(t) > 0, \forall t \in [t_* - \delta, t_* + \delta]$.\\
Now consider the "smooth" function (i.e. infinitely differentiable) given by:
$$ \eta^a(t) = 2 + \sin\left[-\pi/2 + 2 \pi \frac{(t - t_* + \delta)}{2 \delta}\right], \;\; t \in [t_* - \delta, t_* + \delta]$$
Then $\eta^a(t) > 0$ on the same interval that $F_a(t)>0$ and zero elsewhere - hence their product is greater than zero on $[t_* - \delta, t_* + \delta]$ and 
zero elsewhere.  Hence, for:
\begin{align*}
\int_{t_0}^{t_1} F_a(t) \eta^a(t) dt = 0 \;\; \text{for every smooth $\eta^a$ with compact support}
\end{align*}
we must conslude that $F_a(t) = 0$ on $(t_0, t_1)$\\

(b)\\
\begin{align*}
S[q] := \int_{t_0}^{t_1} L(q,\dot q,t)\,dt
\end{align*}
Let $\eta$ be a perturbation of $q$ s.t. $\eta(t_0)=\eta(t_1)=0$. \\
Then:
\begin{align*}
S[q+\epsilon\eta]
  &= \int_{t_0}^{t_1}
     L(q+\epsilon\eta,\dot q+\epsilon\dot\eta,t)\,dt,\\
\Rightarrow\quad
\frac{dS}{d\epsilon}
  &= \int_{t_0}^{t_1}
     \frac{d}{d\epsilon}
     L(q+\epsilon\eta,\dot q+\epsilon\dot\eta,t)\,dt.
     \tag{*}
\end{align*}
Now
\begin{align*}
\frac{d}{d\epsilon}
L(q+\epsilon\eta,\dot q+\epsilon\dot\eta,t)
=
\frac{\partial}{\partial q}L(q+\epsilon\eta,\dot q+\epsilon\dot\eta,t) \eta
+
\frac{\partial}{\partial\dot q}L(q+\epsilon\eta,\dot q+\epsilon\dot\eta,t) \dot\eta.
\end{align*}
So, (*) becomes:
\begin{align*}
\frac{dS}{d\epsilon}
&=
\int_{t_0}^{t_1}
\left(\frac{\partial}{\partial q}L(q+\epsilon\eta,\dot q+\epsilon\dot\eta,t) \eta
+
\frac{\partial}{\partial\dot q}
L(q+\epsilon\eta,\dot q+\epsilon\dot\eta,t)\dot\eta
\right) \\
\implies \left. \frac{dS}{d\epsilon} \right|_{\epsilon=0} &=
\int_{t_0}^{t_1}
\left(\frac{\partial}{\partial q}L(q,\dot q,t) \eta
+
\frac{\partial}{\partial\dot q}
L(q,\dot q,t)\dot\eta
\right)
\tag{**}
\end{align*}
Integrating by parts:
\begin{align*}
\int_{t_0}^{t_1}
\frac{\partial L}{\partial\dot q}\dot\eta\,dt
=
\left[
\frac{\partial L}{\partial\dot q}\eta
\right]_{t_0}^{t_1}
-
\int_{t_0}^{t_1}
\frac{d}{dt}
\left(\frac{\partial L}{\partial\dot q}\right)\eta\,dt.
\end{align*}

\newpage
\noindent
Now $\left[\frac{\partial L}{\partial\dot q}\eta
\right]_{t_0}^{t_1} = 0$, since: $\eta(t_0)=\eta(t_1)=0$.
So, (**) becomes:
\begin{align*}
\left.\frac{dS}{d\epsilon}\right|_{\epsilon=0}
&=
\int_{t_0}^{t_1}
\left(
\frac{\partial L}{\partial q}\eta
-
\frac{d}{dt}
\left(\frac{\partial L}{\partial\dot q}\right)\eta \right)
dt\\
&=
\int_{t_0}^{t_1}
\left(
\frac{\partial L}{\partial q}
-
\frac{d}{dt}
\left(\frac{\partial L}{\partial\dot q}\right)
\right)\eta\,dt.
\end{align*}
Now, by assumption:
\begin{align*}
\left.\frac{dS}{d\epsilon}\right|_{\epsilon=0}
=
\delta S
=
\int_{t_0}^{t_1}
\left(
\frac{\partial L}{\partial q}
-
\frac{d}{dt}
\left(\frac{\partial L}{\partial\dot q}\right)
\right)\eta\,dt
=
0.
\end{align*}
Now $\eta$ is an arbitrary ``perturbation'' -- so use
part (a) of this question to conclude that
\begin{align*}
\frac{\partial L}{\partial q}
-
\frac{d}{dt}
\left(\frac{\partial L}{\partial\dot q}\right)
=
0.
\end{align*}
\qed \\

(c) We are given $F=F(q, t)$ such that:
\begin{align*}
L' &= L+\frac{\partial F}{\partial q^a}\dot q^a
       +\frac{\partial F}{\partial t}\\
\Rightarrow\quad
\frac{\partial L'}{\partial q^a}
&= \frac{\partial L}{\partial q^a}
 +\frac{\partial^2F}{\partial(q^a)^2}\dot q^a
 +\frac{\partial^2 F}{\partial q^a \partial t}.
\end{align*}
We also have that:
\begin{align*}
\frac{\partial L'}{\partial\dot q^a}
&= \frac{\partial L}{\partial\dot q^a}
 +\frac{\partial F}{\partial q^a}\\
\Rightarrow\quad
\frac{d}{dt}\left(\frac{\partial L'}{\partial\dot q^a}\right)
&= \frac{d}{dt}\left(\frac{\partial L}{\partial\dot q^a}\right)
 +\frac{d}{dt}\left(\frac{\partial F}{\partial q^a}\right) \tag{***}
\end{align*}
For the last term in (***):
\begin{align*}
\frac{d}{dt}\left(\frac{\partial F}{\partial q^a}\right)
&= \frac{\partial^2F}{\partial(q^a)^2}\dot q^a
 +\frac{\partial}{\partial t}
  \left(\frac{\partial F}{\partial q^a}\right) \\
&= \frac{\partial^2F}{\partial(q^a)^2}\dot q^a
 +\frac{\partial^2 F}{\partial t \partial q^a}
\end{align*}
Now the order of the 2nd order partials is not relevant (as $F$ is smooth) and hence:
\begin{align*}
\frac{d}{dt}\left(\frac{\partial L'}{\partial\dot q^a}\right)
&= \boxed{\textcolor{red}{
   \frac{d}{dt}\left(\frac{\partial L}{\partial\dot q^a}\right)}}
 +\frac{\partial^2F}{\partial(q^a)^2}\dot q^a
 +\frac{\partial^2F}{\partial t\,\partial q^a}\\
&= \boxed{\textcolor{red}{\frac{\partial L}{\partial q^a}}}
 +\frac{\partial^2F}{\partial(q^a)^2}\dot q^a
 +\frac{\partial^2F}{\partial q^a\,\partial t} \\
&= \frac{\partial L'}{\partial q^a}
\end{align*}
The middle line follows as $L$ satisfies the Langrange equation and the two terms in red boxes
are equal. \\

L=\frac12 m\dot q^2-\frac12 kq^2

E_L
=
\dot q\frac{\partial L}{\partial\dot q}-L

and verify directly that \(\dot E_L=0\) on solutions of the Euler--Lagrange
equation.

\item[(b)] For the plane pendulum of length \(\ell\) use the angle \(\theta\)
(measured from the downward vertical) to derive
\begin{align*}
\ddot\theta+\frac{g}{\ell}\sin\theta=0.
\end{align*}
Determine the small-oscillation frequency, and show that energy conservation
reduces the problem to the quadrature
\begin{align*}
\dot\theta^2

\frac{2g}{\ell}

\right),
\end{align*}
where \(0<\theta_0<\pi\) is the amplitude of a libration.
Deduce the exact period as an integral and check that it tends to
\(2\pi\sqrt{\ell/g}\) as \(\theta_0\to0\).

\item[(c)] Treat the same pendulum in \(\mathbb{R}^2\) with the constraint
\begin{align*}
G(x,y)=x^2+y^2-\ell^2=0
\end{align*}
and
\begin{align*}
L=\frac12 m(\dot x^2+\dot y^2)-mgy.
\end{align*}
Write down the multiplier equations, and determine \(\lambda\) explicitly
as a function of \((x,y,\dot x,\dot y)\).

\item[(d)] Write the constraint force as
\begin{align*}
\vec F_{\mathrm{constraint}}

\begin{align*}

Show that the signed inward tension is
\begin{align*}
\mathcal{T}
=
m\left(g\cos\theta+\ell\dot\theta^2\right),
\qquad
\left|\vec F_{\mathrm{constraint}}\right|

Explain the distinction between a rigid rod, which can exert compression,

(a) \begin{align*}
L &= \frac{1}{2}m\dot{q}^{\,2}-\frac{1}{2}kq^2\\
\Rightarrow\quad \frac{\partial L}{\partial \dot{q}} &= m\dot{q}\\
\Rightarrow\quad E_L &= \dot{q}\frac{\partial L}{\partial \dot{q}}-L\\
&= \dot{q}(m\dot{q})-\left[\frac{1}{2}m\dot{q}^{\,2}+\frac{1}{2}kq^2\right]\\
&= \frac{1}{2}m\dot{q}^{\,2}+\frac{1}{2}kq^2
\end{align*}
But this is The total energy of The System - which
for an isolated system must be constant.  Indeed:
\begin{align*}
\dot{E}_L &= m \dot{q} \ddot{q} + kq \dot{q} \\
&= \dot{q} [m \ddot{q} + kq] \\
&= 0 \;\;\;\;\;\;\;\;\;\;\;\;\;\;\;\;\; \text{(from E-L equations)}
\end{align*}
(b) For The pendulum we have:
\begin{align*}
\text{k.e. of pendulum} &= \frac{1}{2}mv^2
= \frac{1}{2}ml^2\dot{\theta}^2\\
\text{p.e. of pendulum} &= mlg(1-\cos\theta)
\end{align*}

The Lagrangian is:
\begin{align*}
L=\frac{1}{2}ml^2\dot{\theta}^2-mlg(1-\cos\theta)
\end{align*}
Lagrange's dynamical eqn is then:
\begin{align*}
\frac{d}{dt}\frac{\partial L}{\partial\dot{\theta}}
&=\frac{\partial L}{\partial\theta}\\
\Rightarrow\quad
\frac{d}{dt}\left(ml^2\dot{\theta}\right)
&=-mlg\sin\theta\\
\Rightarrow\quad
ml^2\ddot{\theta}
&=-mlg\sin\theta\\
\Rightarrow\quad
\ddot{\theta}+\frac{g}{l}\sin\theta
&=0 \tag{2.1}
\end{align*}
For \textbf{small oscillations} $\theta$ is small $\implies \sin(\theta) = \theta$, then (2.1) becomes:
\begin{align*}
\ddot{\theta} = - \frac{g}{l}\theta \tag{2.2}
\end{align*}
Under such motion, the frequency is given by: $\omega = \sqrt{\frac{g}{l}}$.\\
\\
For energy conservation - given by $E_L$, we have that:
\begin{align*}
E_L &= \dot{\theta}\frac{\partial L}{\partial\dot{\theta}}-L\\
&= \dot{\theta}\left[ml^2\dot{\theta}\right]-\left[\frac{1}{2}ml^2\dot{\theta}^2-mlg(1-\cos\theta)\right]\\
&= \frac{1}{2}ml^2\dot{\theta}^2+mlg(1-\cos\theta).\tag{2.3}
\end{align*}
We know that at $\theta=\theta_0$ (max amplitude) all energy is potential energy (hence $\dot{\theta}=0$). Hence (2.3) becomes
\begin{align*}
mgl(1-\cos\theta_0)&=\frac{1}{2}ml^2\dot{\theta}^2+mlg(1-\cos\theta)\\
\Rightarrow\quad \frac{1}{2}ml^2\dot{\theta}^2&=mlg(\cos\theta-\cos\theta_0)\\
\Rightarrow\quad \dot{\theta}^2&=\frac{2g}{l}(\cos\theta-\cos\theta_0) \tag{2.4}
\end{align*}
\\
Here we consider determining the period of motion from (2.4). 
\begin{align*}
\dot{\theta}&=\sqrt{\frac{2g}{L}(\cos\theta-\cos\theta_0)}\\
\Rightarrow\quad \sqrt{\frac{L}{2g}}\frac{d\theta}{\sqrt{\cos\theta-\cos\theta_0}} & = dt \\
\Rightarrow\quad\sqrt{\frac{L}{2g}} \int_{-\theta_0}^{\theta_0} \frac{d\theta}{\sqrt{\cos\theta-\cos\theta_0}} & = \int_{0}^\frac{T}{2} dt \tag{2.5}
\end{align*}
Where in (2.5) I am integrating over 1/2 period of oscillation, $T/2$, when the pendulum swings from
$-\theta_0$ to $\theta_0$.

\sqrt{\frac{L}{2g}}\int_{-\theta_0}^{\theta_0}

\Rightarrow\quad \sqrt{\frac{L}{g}}\int_{-\theta_0}^{\theta_0}

\end{align*}

Substituting this into (2.6) gives:
\begin{align*}
T = 2\pi\sqrt{\frac{L}{g}}
\end{align*}
